    const DEFAULT_LYRICS_TEMPLATE = ["Intro", "A段主歌", "B段副歌", "桥段", "结尾"]

      .map((section) => `[${section}]`)

      .join("\n\n");

    const LEVEL_LABELS = {

      1: "很不熟练",

      2: "可以顺下来",

      3: "可以练摊儿",

      4: "可以去tribute演"

    };

    const DEFAULT_BAND_SETUP = {

      vocalists: ["米粉儿"],

      rhythmGuitar: "坤哥",

      leadGuitar: "沈总",

      bass: "",

      drums: "谢老师",

      keyboard: "小京"

    };

    const BAND_SCENES = {

      withDrums: "有鼓",

      withoutDrums: "无鼓"

    };


    const state = {

      songs: loadSongs(),

      selectedSongId: null,

      search: "",

      mobileView: "list",

      editingBandScene: "withDrums",

      editingBandSetupScenes: createDefaultBandSetupScenes()

    };


    const elements = {

      songList: document.getElementById("song-list"),

      songCount: document.getElementById("song-count"),

      songSearch: document.getElementById("song-search"),

      emptyState: document.getElementById("empty-state"),

      detailView: document.getElementById("song-detail-view"),

      detailContent: document.getElementById("detail-content"),

      songListView: document.getElementById("song-list-view"),

      addSongBtn: document.getElementById("add-song-btn"),

      backBtn: document.getElementById("back-btn"),

      songModal: document.getElementById("song-modal"),

      songForm: document.getElementById("song-form")

    };


    bootstrap();


    function bootstrap() {

      if (state.songs.length > 0) {

        state.selectedSongId = state.songs[0].id;

      }


      bindEvents();

      render();

    }


    function bindEvents() {

      elements.addSongBtn.addEventListener("click", () => openSongModal());

      elements.backBtn.addEventListener("click", () => {

        state.mobileView = "list";

        syncMobileLayout();

      });


      elements.songSearch.addEventListener("input", (event) => {

        state.search = event.target.value.trim().toLowerCase();

        renderSongList();

      });


      elements.songForm.addEventListener("submit", handleSongSubmit);

      bindVocalistPicker();

      bindBandSceneSwitch();


      document.querySelectorAll("[data-close]").forEach((button) => {

        button.addEventListener("click", () => closeModal(button.dataset.close));

      });


      [elements.songModal].forEach((modal) => {

        modal.addEventListener("click", (event) => {

          if (event.target === modal) {

            closeModal(modal.id);

          }

        });

      });


      window.addEventListener("resize", syncMobileLayout);

    }


    function render() {

      renderSongList();

      renderSongDetail();

      syncMobileLayout();

    }


    function renderSongList() {

      const filteredSongs = getFilteredSongs();

      elements.songCount.textContent = `共 ${filteredSongs.length} 首曲目`;

      elements.emptyState.classList.toggle("hidden", state.songs.length !== 0);

      elements.songList.innerHTML = "";


      if (filteredSongs.length === 0 && state.songs.length > 0) {

        elements.songList.innerHTML = `<div class="detail-empty"><p>没有匹配的曲目。</p></div>`;

        return;

      }


      filteredSongs.forEach((song) => {

        const card = document.createElement("article");

        card.className = `song-card${song.id === state.selectedSongId ? " active" : ""}`;

        card.innerHTML = `

          <div class="record-card-head">

            <div class="song-title-row">

              <h3>${escapeHtml(song.title)}</h3>

              <button class="ghost-btn compact-btn" data-action="view-song" data-id="${song.id}">查看</button>

            </div>

            <span class="star-row">${escapeHtml(renderLevel(song.level))}</span>

          </div>

          <div class="song-meta">

            <span class="tag">调号 ${escapeHtml(song.key || "未填写")}</span>

            ${song.chartLink ? `<span class="tag">有谱子链接</span>` : ""}

            ${song.chordProgression || song.lyrics ? `<span class="tag">有歌词与和弦</span>` : ""}

          </div>

        `;

        elements.songList.appendChild(card);

      });


      elements.songList.querySelectorAll("[data-action]").forEach((button) => {

        button.addEventListener("click", handleSongActions);

      });

    }


    function renderSongDetail() {

      const song = state.songs.find((item) => item.id === state.selectedSongId);


      if (!song) {

        elements.detailContent.innerHTML = `

          <div class="detail-empty">

            <div>

              <h3>选择一首曲目查看详情</h3>

              <p>你可以从左侧列表进入，或先新增曲目。</p>

            </div>

          </div>

        `;

        return;

      }


      const chartLink = song.chartLink

        ? `<a class="text-link" href="${escapeAttribute(song.chartLink)}" target="_blank" rel="noreferrer">${escapeHtml(song.chartLink)}</a>`

        : '<span class="muted">未填写</span>';

      const chordsPreview = song.chordProgression

        ? `<pre class="lyrics-preview">${escapeHtml(song.chordProgression)}</pre>`

        : `<div class="detail-empty compact-empty"><p>还没有填写和弦走向。</p></div>`;

      const lyricsPreview = song.lyrics

        ? `<pre class="lyrics-preview">${escapeHtml(song.lyrics)}</pre>`

        : `<div class="detail-empty compact-empty"><p>还没有填写歌词。</p></div>`;


      elements.detailView.classList.remove("hidden");

      elements.detailContent.innerHTML = `

        <section class="summary-card">

          <div class="detail-header">

            <div>

              <h2>${escapeHtml(song.title)}</h2>

              <p class="muted">调号 ${escapeHtml(song.key || "未填写")} · 熟练度 ${escapeHtml(renderLevel(song.level))}</p>

            </div>

            <div class="detail-actions">

              <button class="ghost-btn" data-action="edit-song" data-id="${song.id}">编辑</button>

              <button class="danger-btn" data-action="delete-song" data-id="${song.id}">删除</button>

            </div>

          </div>

        </section>

        <section class="summary-card info-grid">

          <div>

            <p class="info-label">曲名</p>

            <p class="info-value">${escapeHtml(song.title)}</p>

          </div>

          <div>

            <p class="info-label">调号</p>

            <p class="info-value">${escapeHtml(song.key || "未填写")}</p>

          </div>

          <div>

            <p class="info-label">熟练度</p>

            <p class="info-value">${escapeHtml(renderLevel(song.level))}</p>

          </div>

          <div>

            <p class="info-label">谱子链接</p>

            <div class="info-value">${chartLink}</div>

          </div>

          <div class="full-span">

            <p class="info-label">乐队成员配置</p>

            <div class="band-setup-preview">

              ${renderBandSetupGroups(song.bandSetup)}

            </div>

          </div>

        </section>

        <section class="summary-card">

          <div class="detail-header">

            <div>

              <h2>歌词与和弦</h2>

              <p class="muted">拆分记录和弦走向与歌词段落，保留换行并使用等宽字体。</p>

            </div>

            <button class="ghost-btn" data-action="edit-lyrics" data-song-id="${song.id}">编辑歌词与和弦</button>

          </div>

          <div class="lyrics-section">

            <div id="lyrics-preview-wrap" class="lyrics-preview-grid">

              <div>

                <p class="info-label">和弦走向</p>

                ${chordsPreview}

              </div>

              <div>

                <p class="info-label">歌词</p>

                ${lyricsPreview}

              </div>

            </div>

            <div class="lyrics-editor hidden" id="lyrics-editor-wrap">

              <div class="lyrics-preview-grid">

                <label>

                  <span>和弦走向</span>

                  <textarea id="chords-input" rows="8" class="lyrics-textarea" placeholder="例如：&#10;[Intro]  C   G   Am   F&#10;[Verse]  C   G   F   C">${escapeHtml(song.chordProgression || "")}</textarea>

                </label>

                <label>

                  <span>歌词</span>

                  <textarea id="lyrics-input" rows="12" class="lyrics-textarea" placeholder="默认已帮你放好段落模板">${escapeHtml(song.lyrics || DEFAULT_LYRICS_TEMPLATE)}</textarea>

                </label>

              </div>

              <div class="form-actions">

                <button class="ghost-btn" data-action="cancel-lyrics">取消</button>

                <button class="primary-btn" data-action="save-lyrics" data-song-id="${song.id}">保存歌词与和弦</button>

              </div>

            </div>

          </div>

        </section>

      `;


      elements.detailContent.querySelectorAll("[data-action]").forEach((button) => {

        button.addEventListener("click", handleSongActions);

      });

    }


    function handleSongActions(event) {

      const { action, id, songId } = event.currentTarget.dataset;


      if (action === "view-song") {

        state.selectedSongId = id;

        if (window.innerWidth <= 900) {

          state.mobileView = "detail";

        }

        render();

        return;

      }


      if (action === "edit-song") {

        const song = state.songs.find((item) => item.id === id);

        openSongModal(song);

        return;

      }


      if (action === "delete-song") {

        deleteSong(id);

        return;

      }


      if (action === "edit-lyrics") {

        toggleLyricsEditor(true);

        return;

      }


      if (action === "cancel-lyrics") {

        toggleLyricsEditor(false);

        return;

      }


      if (action === "save-lyrics") {

        saveLyrics(songId);

        return;

      }

    }


    function handleSongSubmit(event) {

      event.preventDefault();

      const id = document.getElementById("song-id").value || createId();

      const existingSong = state.songs.find((item) => item.id === id);

      const song = {

        id,

        title: document.getElementById("song-title").value.trim(),

        key: document.getElementById("song-key").value.trim(),

        level: Number(document.getElementById("song-level").value),

        chartLink: document.getElementById("song-chart").value.trim(),

        bandSetup: readBandSetupFromSongForm(state.editingBandSetupScenes),

        chordProgression: existingSong ? existingSong.chordProgression || "" : "",

        lyrics: existingSong ? existingSong.lyrics || DEFAULT_LYRICS_TEMPLATE : DEFAULT_LYRICS_TEMPLATE

      };


      if (!song.title) {

        return;

      }


      if (existingSong) {

        Object.assign(existingSong, song);

      } else {

        state.songs.unshift(song);

        state.selectedSongId = song.id;

        state.mobileView = "detail";

      }


      persist();

      closeModal("song-modal");

      render();

    }


    function openSongModal(song) {

      document.getElementById("song-modal-title").textContent = song ? "编辑曲目" : "新增曲目";

      document.getElementById("song-id").value = song?.id || "";

      document.getElementById("song-title").value = song?.title || "";

      document.getElementById("song-level").value = String(normalizeLevel(song?.level || 2));

      document.getElementById("song-chart").value = song?.chartLink || "";

      document.getElementById("song-key").value = song?.key || "";

      state.editingBandScene = "withDrums";

      state.editingBandSetupScenes = normalizeBandSetupScenes(song?.bandSetup || createDefaultBandSetupScenes());

      applyBandSetupToSongForm(state.editingBandSetupScenes);


      openModal("song-modal");

      window.setTimeout(() => {

        document.getElementById("song-title").focus();

      }, 30);

    }


    function deleteSong(songId) {

      const song = state.songs.find((item) => item.id === songId);

      const confirmed = window.confirm(`确认删除曲目“${song?.title || ""}”？`);

      if (!confirmed) {

        return;

      }


      state.songs = state.songs.filter((item) => item.id !== songId);

      if (state.selectedSongId === songId) {

        state.selectedSongId = state.songs[0]?.id || null;

        state.mobileView = state.selectedSongId ? "detail" : "list";

      }

      persist();

      render();

    }


    function openModal(modalId) {

      const modal = document.getElementById(modalId);

      modal.classList.remove("hidden");

      modal.setAttribute("aria-hidden", "false");

    }


    function closeModal(modalId) {

      const modal = document.getElementById(modalId);

      modal.classList.add("hidden");

      modal.setAttribute("aria-hidden", "true");

      if (modalId === "song-modal") {

        state.editingBandScene = "withDrums";

        state.editingBandSetupScenes = createDefaultBandSetupScenes();

        elements.songForm.reset();

        document.getElementById("song-id").value = "";

      }

    }


    function getFilteredSongs() {

      if (!state.search) {

        return state.songs;

      }


      return state.songs.filter((song) => {

        const haystack = `${song.title} ${song.key}`.toLowerCase();

        return haystack.includes(state.search);

      });

    }


    function loadSongs() {

      try {

        const stored = localStorage.getItem(STORAGE_KEY);

        if (!stored) {

          return [];

        }


        const parsed = JSON.parse(stored);

        if (!Array.isArray(parsed)) {

          return [];

        }


        return parsed.map((song) => ({

          id: song.id || createId(),

          title: song.title || "",

          key: song.key || "",

          level: normalizeLevel(song.level),

          chartLink: song.chartLink || "",

          bandSetup: normalizeBandSetupScenes(song.bandSetup),

          chordProgression: song.chordProgression || song.lyricsChords || "",

          lyrics: song.lyrics || (song.lyricsChords ? DEFAULT_LYRICS_TEMPLATE : "")

        }));

      } catch (error) {

        console.error("Failed to load songs", error);

        return [];

      }

    }


    function persist() {

      localStorage.setItem(STORAGE_KEY, JSON.stringify(state.songs));

    }


    function syncMobileLayout() {

      if (window.innerWidth > 900) {

        elements.songListView.classList.remove("hidden");

        elements.detailView.classList.remove("hidden");

        return;

      }


      const showDetail = state.mobileView === "detail" && Boolean(state.selectedSongId);

      elements.songListView.classList.toggle("hidden", showDetail);

      elements.detailView.classList.toggle("hidden", !showDetail);

    }


    function toggleLyricsEditor(shouldOpen) {

      const editorWrap = document.getElementById("lyrics-editor-wrap");

      const previewWrap = document.getElementById("lyrics-preview-wrap");

      if (!editorWrap || !previewWrap) {

        return;

      }


      editorWrap.classList.toggle("hidden", !shouldOpen);

      previewWrap.classList.toggle("hidden", shouldOpen);


      if (shouldOpen) {

        const input = document.getElementById("chords-input");

        if (input) {

          input.focus();

          input.setSelectionRange(input.value.length, input.value.length);

        }

      }

    }


    function saveLyrics(songId) {

      const song = state.songs.find((item) => item.id === songId);

      const chordsInput = document.getElementById("chords-input");

      const input = document.getElementById("lyrics-input");

      if (!song || !input || !chordsInput) {

        return;

      }


      song.chordProgression = chordsInput.value.replace(/\r\n/g, "\n").trim();

      song.lyrics = input.value.replace(/\r\n/g, "\n");

      persist();

      renderSongDetail();

    }


    function readBandSetupFromSongForm(existingBandSetup = state.editingBandSetupScenes) {

      const scenes = normalizeBandSetupScenes(existingBandSetup);

      scenes[state.editingBandScene] = {

        vocalists: Array.from(document.querySelectorAll('input[name="song-vocalists"]:checked')).map((input) => input.value),

        rhythmGuitar: getSelectValue("song-rhythmGuitar"),

        leadGuitar: getSelectValue("song-leadGuitar"),

        bass: getSelectValue("song-bass"),

        drums: getSelectValue("song-drums"),

        keyboard: getSelectValue("song-keyboard")

      };

      return normalizeBandSetupScenes(scenes);

    }


    function applyBandSetupToSongForm(bandSetupScenes) {

      const normalizedScenes = normalizeBandSetupScenes(bandSetupScenes);

      const normalized = normalizedScenes[state.editingBandScene];

      document.querySelectorAll('input[name="song-vocalists"]').forEach((input) => {

        input.checked = normalized.vocalists.includes(input.value);

      });

      updateVocalistPickerLabel();

      setSelectValue("song-rhythmGuitar", normalized.rhythmGuitar);

      setSelectValue("song-leadGuitar", normalized.leadGuitar);

      setSelectValue("song-bass", normalized.bass);

      setSelectValue("song-drums", normalized.drums);

      setSelectValue("song-keyboard", normalized.keyboard);

      syncBandSceneButtons();

      syncDrumOptions();

    }


    function normalizeBandSetup(bandSetup, allowNoDrums = false) {

      if (bandSetup) {

        return {

          vocalists: Array.isArray(bandSetup.vocalists) ? bandSetup.vocalists : [...DEFAULT_BAND_SETUP.vocalists],

          rhythmGuitar: bandSetup.rhythmGuitar || DEFAULT_BAND_SETUP.rhythmGuitar,

          leadGuitar: bandSetup.leadGuitar || DEFAULT_BAND_SETUP.leadGuitar,

          bass: typeof bandSetup.bass === "string" ? bandSetup.bass : DEFAULT_BAND_SETUP.bass,

          drums: allowNoDrums ? (typeof bandSetup.drums === "string" ? bandSetup.drums : "") : (bandSetup.drums || DEFAULT_BAND_SETUP.drums),

          keyboard: bandSetup.keyboard || DEFAULT_BAND_SETUP.keyboard

        };

      }

      return {

        ...DEFAULT_BAND_SETUP,

        vocalists: [...DEFAULT_BAND_SETUP.vocalists],

        drums: allowNoDrums ? "" : DEFAULT_BAND_SETUP.drums

      };

    }


    function normalizeBandSetupScenes(bandSetupScenes) {

      if (bandSetupScenes?.withDrums || bandSetupScenes?.withoutDrums) {

        return {

          withDrums: normalizeBandSetup(bandSetupScenes.withDrums, false),

          withoutDrums: normalizeBandSetup(bandSetupScenes.withoutDrums, true)

        };

      }


      const legacy = normalizeBandSetup(bandSetupScenes, false);

      return {

        withDrums: legacy,

        withoutDrums: {

          ...legacy,

          vocalists: [...legacy.vocalists],

          drums: ""

        }

      };

    }


    function createDefaultBandSetupScenes() {

      return normalizeBandSetupScenes(null);

    }


    function renderBandSetupGroups(bandSetupScenes) {

      const normalizedScenes = normalizeBandSetupScenes(bandSetupScenes);

      return Object.entries(BAND_SCENES).map(([sceneKey, sceneLabel]) => `

        <div class="band-scene-group">

          <p class="info-label">${sceneLabel}</p>

          <div class="member-list">${renderBandSetupTags(normalizedScenes[sceneKey])}</div>

        </div>

      `).join("");

    }


    function renderBandSetupTags(bandSetup) {

      const normalized = normalizeBandSetup(bandSetup, true);

      const entries = [

        `主唱：${normalized.vocalists.length ? normalized.vocalists.join("、") : "无"}`,

        `节奏吉他：${normalized.rhythmGuitar || "无"}`,

        `旋律吉他：${normalized.leadGuitar || "无"}`,

        `贝斯：${normalized.bass || "无"}`,

        `鼓：${normalized.drums || "无"}`,

        `键盘：${normalized.keyboard || "无"}`

      ];

      return entries.map((item) => `<span class="member-tag">${escapeHtml(item)}</span>`).join("");

    }


    function getSelectValue(id) {

      const element = document.getElementById(id);

      return element ? element.value : "";

    }


    function setSelectValue(id, value) {

      const element = document.getElementById(id);

      if (element) {

        element.value = value;

      }

    }


    function bindVocalistPicker() {

      const toggle = document.getElementById("song-vocalists-toggle");

      const panel = document.getElementById("song-vocalists-panel");

      if (!toggle || !panel) {

        return;

      }


      toggle.addEventListener("click", () => {

        const isOpen = !panel.classList.contains("hidden");

        panel.classList.toggle("hidden", isOpen);

        toggle.setAttribute("aria-expanded", String(!isOpen));

      });


      document.querySelectorAll('input[name="song-vocalists"]').forEach((input) => {

        input.addEventListener("change", updateVocalistPickerLabel);

      });


      document.addEventListener("click", (event) => {

        if (!panel.contains(event.target) && !toggle.contains(event.target)) {

          panel.classList.add("hidden");

          toggle.setAttribute("aria-expanded", "false");

        }

      });


      updateVocalistPickerLabel();

    }


    function updateVocalistPickerLabel() {

      const toggle = document.getElementById("song-vocalists-toggle");

      const text = toggle?.querySelector(".picker-toggle-text");

      if (!toggle) {

        return;

      }


      const selected = Array.from(document.querySelectorAll('input[name="song-vocalists"]:checked')).map((input) => input.value);

      if (text) {

        text.textContent = selected.length ? selected.join("、") : "请选择主唱";

      }

    }


    function bindBandSceneSwitch() {

      document.querySelectorAll(".scenario-btn").forEach((button) => {

        button.addEventListener("click", () => {

          const nextScene = button.dataset.scene;

          if (!nextScene || nextScene === state.editingBandScene) {

            return;

          }


          state.editingBandSetupScenes = readBandSetupFromSongForm(state.editingBandSetupScenes);

          state.editingBandScene = nextScene;

          applyBandSetupToSongForm(state.editingBandSetupScenes);

        });

      });

    }


    function syncBandSceneButtons() {

      document.querySelectorAll(".scenario-btn").forEach((button) => {

        button.classList.toggle("active", button.dataset.scene === state.editingBandScene);

      });

    }


    function syncDrumOptions() {

      const drumSelect = document.getElementById("song-drums");

      if (!drumSelect) {

        return;

      }


      const allowEmpty = state.editingBandScene === "withoutDrums";

      const emptyOption = drumSelect.querySelector('option[value=""]');

      if (emptyOption) {

        emptyOption.hidden = !allowEmpty;

      }


      if (!allowEmpty && !drumSelect.value) {

        drumSelect.value = DEFAULT_BAND_SETUP.drums;

      }

    }


    function createId() {

      return `${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}`;

    }


    function normalizeLevel(level) {

      const numeric = Number(level);

      if (numeric >= 5) {

        return 4;

      }

      if (numeric >= 4) {

        return 3;

      }

      if (numeric >= 2) {

        return 2;

      }

      return 1;

    }


    function renderLevel(level) {

      return LEVEL_LABELS[normalizeLevel(level)] || LEVEL_LABELS[1];

    }


    function escapeHtml(value) {

      return String(value)

        .replaceAll("&", "&amp;")

        .replaceAll("<", "&lt;")

        .replaceAll(">", "&gt;")

        .replaceAll('"', "&quot;")

        .replaceAll("'", "&#39;");

    }


    function escapeAttribute(value) {

      return escapeHtml(value);

    }
